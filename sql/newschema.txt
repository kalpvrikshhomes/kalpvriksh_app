-- Drop public schema objects safely
DO $$
DECLARE
  r RECORD;
BEGIN
  FOR r IN (
    SELECT tablename FROM pg_tables WHERE schemaname = 'public'
  ) LOOP
    EXECUTE 'DROP TABLE IF EXISTS public.' || quote_ident(r.tablename) || ' CASCADE';
  END LOOP;

  FOR r IN (
    SELECT typname FROM pg_type
    JOIN pg_namespace n ON n.oid = pg_type.typnamespace
    WHERE n.nspname = 'public'
  ) LOOP
    EXECUTE 'DROP TYPE IF EXISTS public.' || quote_ident(r.typname) || ' CASCADE';
  END LOOP;
END $$;

⚙️ 03_functions.sql (AUTOMATION)
Inventory update
CREATE OR REPLACE FUNCTION update_inventory_quantity()
RETURNS trigger
LANGUAGE plpgsql AS $$
BEGIN
  UPDATE inventory_items
  SET total_quantity = total_quantity + NEW.quantity_change
  WHERE id = NEW.inventory_item_id
    AND total_quantity + NEW.quantity_change >= 0;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Insufficient inventory stock';
  END IF;

  RETURN NEW;
END;
$$;

Material issue → inventory history
CREATE OR REPLACE FUNCTION log_material_issue()
RETURNS trigger
LANGUAGE plpgsql AS $$
BEGIN
  INSERT INTO inventory_history (
    inventory_item_id,
    quantity_change,
    reason,
    related_project_id,
    created_by
  )
  VALUES (
    NEW.inventory_item_id,
    -NEW.quantity,
    'issued_to_project',
    NEW.project_id,
    NEW.issued_by
  );

  RETURN NEW;
END;
$$;

Close project → auto profit
CREATE OR REPLACE FUNCTION close_project_finance()
RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE
  material_cost numeric;
  worker_cost numeric;
BEGIN
  SELECT SUM(quantity * rate_at_issue)
  INTO material_cost
  FROM material_issues
  WHERE project_id = NEW.id;

  SELECT SUM(amount)
  INTO worker_cost
  FROM worker_payments
  WHERE project_id = NEW.id;

  INSERT INTO project_finance (
    project_id,
    material_cost,
    worker_cost,
    profit
  )
  VALUES (
    NEW.id,
    COALESCE(material_cost,0),
    COALESCE(worker_cost,0),
    NEW.project_value - COALESCE(material_cost,0) - COALESCE(worker_cost,0)
  );

  RETURN NEW;
END;
$$;

🔁 04_triggers.sql
CREATE TRIGGER trg_inventory_history
AFTER INSERT ON inventory_history
FOR EACH ROW
EXECUTE FUNCTION update_inventory_quantity();

CREATE TRIGGER trg_material_issue
AFTER INSERT ON material_issues
FOR EACH ROW
EXECUTE FUNCTION log_material_issue();

CREATE TRIGGER trg_project_close
AFTER UPDATE OF status ON projects
FOR EACH ROW
WHEN (NEW.status = 'completed')
EXECUTE FUNCTION close_project_finance();
